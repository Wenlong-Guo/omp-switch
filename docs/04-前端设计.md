const providerSchema = z.object({
  id: z.string().min(1, "Provider ID 不能为空"),

## Tauri API 封装

```typescript
// lib/tauri-api.ts
import { invoke } from "@tauri-apps/api/core";
import { listen } from "@tauri-apps/api/event";

export async function invokeCommand<T>(cmd: string, payload?: unknown): Promise<T> {
  try {
    return await invoke<T>(cmd, payload);
  } catch (error) {
    console.error(`IPC 调用失败 [${cmd}]:`, error);
    throw error;
  }
}

// 监听后端文件变更事件
export function onConfigChanged(callback: () => void) {
  return listen("config:changed", callback);
}
```

## 事件流

```
用户点击「保存 Provider」
  → ProviderForm.onSubmit → providerStore.saveProvider(config)
  → invokeCommand("save_provider", config)
  → Rust 写入 SQLite + models.yml
  → 返回成功
  → 前端 Toast 提示 + 刷新列表
  
用户外部编辑 models.yml
  → FileWatcher 检测 → 发送 config:changed 事件
  → 前端 onConfigChanged 监听 → providerStore.fetchProviders()
  → UI 自动刷新
```
